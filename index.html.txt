    script
         CONFIGURACIÓN INICIAL
        const categories = [
            { n 'Lácteos', d 15, e '🥛' }, { n 'Panadería', d 5, e '🍞' }, { n 'Carnes', d 3, e '🥩' },
            { n 'Frutas', d 4, e '🍎' }, { n 'Bebidas', d 30, e '🥤' }, { n 'Almacén', d 180, e '🥫' }
        ];

        let database = JSON.parse(localStorage.getItem('vp_db'))  [];
        let memory = JSON.parse(localStorage.getItem('vp_memory'))  {};  Aquí vive el autocompletado

         INICIALIZAR ESCÁNER
        const codeReader = new ZXing.BrowserMultiFormatReader();
        codeReader.decodeFromVideoDevice(null, 'video', (result, err) = {
            if (result) {
                const code = result.text;
                document.getElementById('barcode-input').value = code;
                
                 --- AQUÍ ESTÁ EL ARREGLO DE LA MEMORIA ---
                if (memory[code]) {
                    document.getElementById('product-name').value = memory[code].name;
                    selectCategory(memory[code].catIndex);
                     Efecto visual de que encontró algo
                    document.getElementById('product-name').classList.add('bg-green-50');
                    setTimeout(() = document.getElementById('product-name').classList.remove('bg-green-50'), 500);
                }
            }
        });

         GENERAR GRILLA DE CATEGORÍAS
        const grid = document.getElementById('categories-grid');
        let selectedCatIndex = null;

        categories.forEach((cat, index) = {
            const div = document.createElement('div');
            div.className = `category-card bg-gray-50 p-2 rounded-lg text-center border-2 border-transparent`;
            div.id = `cat-${index}`;
            div.innerHTML = `div class=text-xl${cat.e}divdiv class=text-[10px]${cat.n}div`;
            div.onclick = () = selectCategory(index);
            grid.appendChild(div);
        });

        function selectCategory(index) {
            document.querySelectorAll('.category-card').forEach(el = el.classList.remove('selected-cat'));
            document.getElementById(`cat-${index}`).classList.add('selected-cat');
            selectedCatIndex = index;
            
             Sugerir fecha
            const dateInput = document.getElementById('expiry-date');
            const targetDate = new Date();
            targetDate.setDate(targetDate.getDate() + categories[index].d);
            dateInput.value = targetDate.toISOString().split('T')[0];
        }

        function saveProduct() {
            const name = document.getElementById('product-name').value;
            const code = document.getElementById('barcode-input').value;
            const date = document.getElementById('expiry-date').value;

            if (!name  !date  !code) return alert(Faltan datos);

            const item = { name, code, date, catIndex selectedCatIndex };
            database.push(item);
            
             Guardar en memoria para la próxima vez
            memory[code] = { name, catIndex selectedCatIndex };
            
            localStorage.setItem('vp_db', JSON.stringify(database));
            localStorage.setItem('vp_memory', JSON.stringify(memory));

            alert(¡Guardado!);
            location.reload();  Recargar para limpiar
        }

        function switchView(view) {
            document.getElementById('view-scan').classList.toggle('hidden', view !== 'scan');
            document.getElementById('view-list').classList.toggle('hidden', view !== 'list');
            renderList();
        }

        function renderList() {
            const container = document.getElementById('inventory-list');
            container.innerHTML = ;
            const hoy = new Date();

            database.sort((a,b) = new Date(a.date) - new Date(b.date)).forEach(item = {
                const vDate = new Date(item.date);
                const diffDays = Math.ceil((vDate - hoy)  (1000  60  60  24));
                
                let statusClass = ;
                if (diffDays = 2) statusClass = status-red;
                else if (diffDays = 7) statusClass = status-yellow;

                container.innerHTML += `
                    div class=bg-white p-4 rounded-xl shadow-sm flex justify-between items-center ${statusClass}
                        div
                            div class=font-bold${item.name}div
                            div class=text-xs text-gray-400${item.code}div
                        div
                        div class=text-right
                            div class=text-sm font-bold ${diffDays = 2  'text-red-500'  'text-gray-600'}${diffDays} díasdiv
                            div class=text-[10px] text-gray-400${item.date}div
                        div
                    div
                `;
            });
        }

        function shareWhatsApp() {
            const urgentes = database.filter(item = {
                const diff = (new Date(item.date) - new Date())  (1000606024);
                return diff = 7;
            });
            
            let msg = 🔥 ¡OFERTAS DE HOY! - Productos por vencernn;
            urgentes.forEach(i = msg += `• ${i.name} (Vence pronto)n`);
            window.open(`httpswa.me{encodeURIComponent(msg)}`);
        }
